\section{Note on data}
\label{sec:data}

\subsection{PPMI features}
 This subsection lists all features found in the PPMI dataset downloaded.
 \cref{tab:features_ppmi} describes each feature while also including their value range
 and whether they were considered for entering the train-validation-test split.
 Note that, to the best of the author's knowledge, 
 there is not an official source explaining the dataset,
 and so the information presented must be taken with a grain of salt.
 For more details on the PPMI dataset, see \cite{Craig2021, Kern2021}.

 \begin{table}[!ht]
    \centering 
    \begin{threeparttable}
    \caption{PPMI features and their meanings.}
    \label{tab:features_ppmi}
    \begin{tabular}{c | c p{3.5cm} p{3.5cm} c}
        \toprule
        Name & Type & Value & Description & Considered \\ 
        \midrule
        \texttt{HudAlphaSample} 
            & identifier & 5104-SL-XXXX, with XXXX a 4-digit number & sample unique identifier & no \\ 
        \texttt{Small RNA-Seq} 
            & categorical & ``Yes'' or ``No'' & whether sample contained scRNA & no \\ 
        \texttt{Long RNA-seq} 
            & categorical & ``Yes'' or ``No'' & whether sample contained lncRNA & no \\ 
        \texttt{PATNO} 
            & identifier & PP-Y, with Y a 4 or 5-digit number & patient identifier & no \\ 
        \texttt{PATNO Visit} 
            & identifier & Y\_CE, where Y is the number from \texttt{PATNO} 
              and CE the \texttt{Clinical Event} 
            & patient visit identifier & no \\ 
        \texttt{QC} 
            & categorical & ``HighQC'' or ``LowQC'' & patient's quality of care & no$^1$ \\ 
        \texttt{PoolAssign} 
            & categorical & ``Sample'' or ``POOL'' & whether sample comes from a patient & no$^2$ \\ 
        \texttt{Phase} 
            & categorical & ``phase1'' or ``phase2'' & from which batch the sample comes from & no \\ 
        \texttt{Disease Status} 
            & diagnostic 
            & ``Idiopathic PD'', ``Healthy Control'', ``Genetic PD'', 
            ``Genetic Unaffected'', ``Prodromal'', or ``SWEDD''$^3$
            & patient diagnosis & no \\ 
        \texttt{Study Arm} 
            & diagnostic 
            & ``PD'', ``HC'', ``GENPD'',  ``GENUN'', 
              ``REGPD'', ``PRODROMA'' or ``SWEDD'' 
            & patient diagnosis & no \\ 
        \bottomrule 
    \end{tabular}
    %
    \begin{tablenotes}
        \item $^1$ Inclusion of QC made correlation calculation unstable.
        \item $^2$ 
            When a sample is POOL, all feature values become POOL.
            Data in the splits considered only patient samples.
        \item $^3$ Scans Without Evidence of Dopaminergic Deficit.
    \end{tablenotes}   
    \end{threeparttable}
\end{table}

            & categorical & ``BL'' or ``VZZ'', with ZZ a 2-digit number & patient visit identifier & no \\

            & categorical & ``MW'', with W a 1 or 2-digit number & month of the patient's visit & no \\ 
        \texttt{Sex} 
            & categorical & ``Female'' or ``Male'' & patient's sex & yes \\ 
        \texttt{Ethnicity} 
            & categorical & ``Not Hispanic or Latino'' or ``Hispanic or Latino'' & patient ethnicity & yes \\

            & categorical 
            & ``White'', ``Black or African American'', ``Multiracial'', 
              ``Asian'', ``Unknown'', ``American Indian or Alaska Native'', ``Other''

            & categorical & ``12-16 years'', ``Greater than 16 years'', ``Less than 12 years'' & patient's education level & yes \\ 
        \texttt{Age (Bin)}
            & categorical & ``under\_55'', ``over\_65'', ``55\_to\_65'' & range within the patient's age is & no \\

            & floating point & positive decimal number & patient's age at consent to participate & yes \\
        \texttt{Age at diagnosis}
            & integer & positive integer & patient's age at diagnosis & no \\
        \texttt{RIN Value}
            & floating point & $[1, 10]$ & RNA integrity number & yes \\
        \texttt{Box}
            & categorical 
            & ``AAAAA-J\_RNA\_B'' or ``J\_RNA\_B'' or ``B'', 
              with AAAAA a 5-digit number and B a 1 or 2-digit number 
            & sample's box & no \\
        \texttt{Position} 
            & integer & positive integer & sample's position within plate & no \\ 
        \texttt{Plate} 
            & categorical & ``pN'', with N a positive integer & sample's plate's identifier & no \\ 
        \bottomrule 
    \end{tabular}
    \end{threeparttable}
